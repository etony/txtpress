# 在文件末尾添加新函数 def extract_mobi_metadata(mobi_path: Path) -> dict: """从 MOBI 文件中提取书籍元数据""" import mobi from mobi.mobi_header import MobiHeader from mobi.mobi_sectioner import Sectionizer # 初始化返回字典 metadata = { 'title': '', 'creator': '', 'publisher': '', 'description': '', 'isbn': '', 'language': '', 'published': '', 'subject': '', 'cover_offset': None } try: # 使用 Sectionizer 读取 MOBI 文件 sect = Sectionizer(str(mobi_path)) mobi_header = MobiHeader(sect, 0) exth_metadata = mobi_header.getMetaData() # 提取元数据字段 metadata['title'] = exth_metadata.get('Title', [''])[0] metadata['creator'] = exth_metadata.get('Creator', [''])[0] metadata['publisher'] = exth_metadata.get('Publisher', [''])[0] metadata['description'] = exth_metadata.get('Description', [''])[0] metadata['isbn'] = exth_metadata.get('ISBN', [''])[0] metadata['language'] = exth_metadata.get('Language', [''])[0] metadata['published'] = exth_metadata.get('Published', [''])[0] metadata['subject'] = exth_metadata.get('Subject', [''])[0] # 提取封面偏移量 if 'CoverOffset' in exth_metadata: metadata['cover_offset'] = int(exth_metadata['CoverOffset'][0]) except Exception as e: logger.error(f'提取 MOBI 元数据失败: {e}') return metadata

# 在 services.py 中添加封面提取函数 def extract_mobi_cover(mobi_path: Path, cover_offset: int) -> Optional[bytes]: """从 MOBI 文件中提取封面图片""" try: sect = Sectionizer(str(mobi_path)) # 封面通常在资源部分 cover_data = sect.loadSection(cover_offset) return cover_data except Exception as e: logger.error(f'提取 MOBI 封面失败: {e}') return None
